\subsection{Derivation of Equations 1.1--1.3}
% TODO (Q1a): derivation starting from R_{e,t+1} = (P_{t+1} + D_{t+1}) / P_t

Throughout, a lowercase letter denotes the log of the corresponding uppercase variable,
e.g.\ $r_{t+1} = \log(R_{t+1})$, $p_t = \log(P_t)$, and $\mathit{cf}_{t+1} = \log(\mathit{CF}_{t+1})$.

\paragraph{Step 1: log return.}
Start from the definition of the gross return,
\begin{equation}
  R_{t+1} = \frac{P_{t+1} + \mathit{CF}_{t+1}}{P_t}.
  \label{eq:q1a_gross_return}
\end{equation}
Taking logs,
\begin{align}
  \log(R_{t+1})
    &= \log(P_{t+1}) - \log(P_t) + \log\!\left(1 + \frac{\mathit{CF}_{t+1}}{P_{t+1}}\right) \nonumber \\
  \Longleftrightarrow \quad
  r_{t+1}
    &= p_{t+1} - p_t + \log\!\left(1 + e^{\,\mathit{cf}_{t+1} - p_{t+1}}\right).
  \label{eq:q1a_log_return}
\end{align}

\paragraph{Step 2: first-order Taylor approximation of the log term.}
Rearranging (\ref{eq:q1a_log_return}) so that the polynomial terms are on the left-hand side
and the log term is on the right-hand side,
\begin{equation}
  r_{t+1} - p_{t+1} + p_t = \log\!\left(1 + e^{\,\mathit{cf}_{t+1} - p_{t+1}}\right).
  \label{eq:q1a_rearranged}
\end{equation}
Let $x_{t+1} = \mathit{cf}_{t+1} - p_{t+1}$. The function $\log(1 + e^{x})$ can be
approximated by a first-order Taylor expansion around a point $a$:
\begin{equation}
  \log\!\left(1 + e^{x}\right)
    \approx \log\!\left(1 + e^{a}\right) + \frac{e^{a}}{1 + e^{a}}\,(x - a),
  \label{eq:q1a_taylor}
\end{equation}
which is accurate for small enough $x - a$. The expansion point $a$ is taken to be the
time average of $\mathit{cf}_{t+1} - p_{t+1}$, which is assumed to converge to a constant,
denoted $a = \overline{\mathit{cf}} - \overline{p}$.

\paragraph{Step 3: substituting the approximation.}
Applying the Taylor approximation (\ref{eq:q1a_taylor}) to the right-hand side of
(\ref{eq:q1a_rearranged}), with $x = \mathit{cf}_{t+1} - p_{t+1}$ and
$a = \overline{\mathit{cf}} - \overline{p}$, and rearranging the terms, we obtain
\begin{equation}
  p_t = \kappa_0 + (1 - \kappa)\,\mathit{cf}_{t+1} - r_{t+1} + \kappa\, p_{t+1},
  \label{eq:q1a_price_recursion}
\end{equation}
where, as in the problem statement,
\begin{equation}
  \kappa = \frac{1}{1 + e^{\,a}}
  \qquad \text{and} \qquad
  \kappa_0 = -\log(\kappa) - (1 - \kappa)\,\log\!\left(\frac{1}{\kappa} - 1\right),
  \label{eq:q1a_kappa_def}
\end{equation}
with $a = \overline{\mathit{cf}} - \overline{p}$ the expansion point of Step 2
(the problem statement writes this average as $\overline{\dpr}$).

\paragraph{Step 4: rewriting in terms of the log cash-flow--price ratio.}
Subtracting $\mathit{cf}_t$ from both sides of (\ref{eq:q1a_price_recursion}) and multiplying
both sides by $-1$, and defining
\begin{equation}
  \dpr_t := \mathit{cf}_t - p_t
  \qquad \text{and} \qquad
  \Delta \mathit{cf}_{t+1} := \mathit{cf}_{t+1} - \mathit{cf}_t,
  \label{eq:q1a_dp_def}
\end{equation}
we obtain
\begin{equation}
  \dpr_t = -\kappa_0 + \kappa\, \dpr_{t+1} + r_{t+1} - \Delta \mathit{cf}_{t+1}.
  \label{eq:q1a_dp_recursion}
\end{equation}

\paragraph{Step 5: forward substitution.}
Equation (\ref{eq:q1a_dp_recursion}) holds at every date, so shifting it one period forward,
\begin{equation}
  \dpr_{t+1} = -\kappa_0 + \kappa\, \dpr_{t+2} + r_{t+2} - \Delta \mathit{cf}_{t+2}.
  \label{eq:q1a_dp_recursion_t1}
\end{equation}
Plugging (\ref{eq:q1a_dp_recursion_t1}) into the $\dpr_{t+1}$ term on the right-hand side of
(\ref{eq:q1a_dp_recursion}) gives
\begin{equation}
  \dpr_t = -\kappa_0 - \kappa\,\kappa_0 + \kappa^{2}\, \dpr_{t+2}
           + r_{t+1} + \kappa\, r_{t+2}
           - \Delta \mathit{cf}_{t+1} - \kappa\, \Delta \mathit{cf}_{t+2}.
  \label{eq:q1a_two_step}
\end{equation}

\paragraph{Step 6: substitution up to horizon $H$.}
Continuing to substitute the future $\dpr$ terms into (\ref{eq:q1a_two_step}) up to
horizon $H$, we get
\begin{equation}
  \dpr_t = -\kappa_0 \sum_{h=0}^{H-1} \kappa^{h}
           + \kappa^{H}\, \dpr_{t+H}
           + \sum_{h=1}^{H} \kappa^{h-1}\, r_{t+h}
           - \sum_{h=1}^{H} \kappa^{h-1}\, \Delta \mathit{cf}_{t+h}.
  \label{eq:q1a_H_step}
\end{equation}

\subsection{Amihud and Hurvich (2004) bias-corrected estimate}
% TODO (Q2c): contrast with Q2b and explain why the two estimates differ (student to write)
Table~\ref{tab:q2c} reports the Amihud and Hurvich (2004) estimate $\hat b^c$ of $b$ in
regression (\ref{eq:q2b_regression}), on the same sample as Table~\ref{tab:q2b}. First,
$D_{t+1}/P_{t+1}$ is regressed on $D_t/P_t$ by OLS, giving $\hat\theta$ and $\hat\rho$, and the
slope is bias-corrected as
$\hat\rho^c = \hat\rho + (1 + 3\hat\rho)/n + 3(1 + 3\hat\rho)/n^2$ with $n = 94$ years.
Second, with $\hat u^c_{t+1} = D_{t+1}/P_{t+1} - (\hat\theta + \hat\rho^c D_t/P_t)$,
\begin{equation}
  \xR_{e,t+1} = a + b\,\frac{D_t}{P_t} + b_u\,\hat u^c_{t+1} + \varepsilon_t
  \label{eq:q2c_regression}
\end{equation}
is estimated by OLS, and $\hat b^c$ is the coefficient on $D_t/P_t$. Its standard error follows
Amihud and Hurvich (2004, Eq.~10),
$\widehat{SE}^c(\hat b^c) = \sqrt{(\hat b_u)^2 \,(1 + 3/n + 9/n^2)^2\, \widehat{\Var}(\hat\rho)
+ \widehat{SE}(\hat b^c)^2}$ with $n = 94$ (code: \texttt{src/q2c.py}).

\subsection{Momentum signal construction and validation}
% TODO (Q3a): interpretation of the results (student to write)

The momentum signal of stock $j$ at the end of month $\tau$ is
\begin{equation}
  \MOM_{j,\tau} = \prod_{m=\tau-11}^{\tau-1} \left(1 + R_{j,m}\right) - 1,
  \label{eq:q3a_mom}
\end{equation}
the cumulative return from the end of month $\tau-12$ to the end of month $\tau-1$, where
$R_{j,m}$ is the CRSP monthly return (\texttt{MthRet}, which includes delisting returns). The
returns are taken from every month of the CRSP file for the same PERMNO, whether or not the
stock is in the universe in those months, and $\MOM_{j,\tau}$ is missing when any of the 11
months has a missing return or no observation.

The signal months are 1963:06--2024:12. The universe at $\tau$ consists of common stocks of
firms incorporated in the United States (CRSP fields \texttt{ShareType = NS},
\texttt{SecurityType = EQTY}, \texttt{SecuritySubType = COM}, \texttt{USIncFlg = Y},
\texttt{IssuerType} $\in$ \{\texttt{ACOR}, \texttt{CORP}\}, the equivalent of share codes 10
and 11) trading on NYSE, Amex or Nasdaq (\texttt{PrimaryExch} $\in$ \{\texttt{N}, \texttt{A},
\texttt{Q}\}, \texttt{ConditionalType = RW}, \texttt{TradingStatusFlg = A}), excluding
utilities ($4900 \le \mathit{SIC} \le 4949$) and financials ($6000 \le \mathit{SIC} \le 6999$)
and requiring non-missing market equity at $\tau$. The SIC code is the CRSP \texttt{SICCD} at
$\tau$; a code of zero is treated as missing and replaced by the most recent earlier non-zero
code of the same PERMNO or, if there is none, by its next later one, and a stock that never
has a non-zero code is kept.

For each month $\tau$, the cross-firm regression
\begin{equation}
  \MOM^{CZ}_{j,\tau} = a_\tau + b_\tau\, \MOM_{j,\tau} + \varepsilon_{j,\tau}
  \label{eq:q3a_regression}
\end{equation}
is estimated by OLS, where $\MOM^{CZ}$ is \texttt{Mom12m} from the CZ Dataset, merged on the
stock identifier and the month $\tau$, using the stocks in the universe at $\tau$ for which
both signals are non-missing (2{,}428{,}767 stock-month observations in 739 months).
Figures~\ref{fig:q3a_intercept}--\ref{fig:q3a_r2} plot $\hat a_\tau$, $\hat b_\tau$ and the
$R^2$ of these regressions, and Table~\ref{tab:q3a} summarizes them (code:
\texttt{src/q3a.py}, \texttt{src/q3\_common.py}).

\subsection{Book-to-market signal construction and validation}
% TODO (Q3b): interpretation of the results (student to write)
% Figures produced by src/q3b.py (common rules in src/q3_common.py)
% from spec/q3b.md and spec/q3_common.md.
The book-to-market signal of stock $j$ for year $t$ is
\begin{equation}
  \BM_{j,t} = \frac{\mathit{BE}_{j,t-1}}{\mathit{ME}_{j,t-1}},
  \label{eq:q3b_bm}
\end{equation}
where $\mathit{BE}_{j,t-1}$ is the book equity from the COMPUSTAT record whose fiscal year
ends in calendar year $t-1$ and $\mathit{ME}_{j,t-1} = |\mathit{PRC}| \times \mathit{SHROUT}$
is the CRSP market equity of the stock in December of year $t-1$ (book equity is in millions
and market equity in thousands of dollars, so the ratio is multiplied by 1000).
$\BM_{j,t}$ is assigned to the months $\tau$ from June of year $t$ to May of year $t+1$, for
$t = 1963, \dots, 2024$, so the signal months are 1963:06--2024:12; the universe at $\tau$ is
the one of Question~3a. $\BM_{j,t}$ is missing when the stock has no CRSP observation or no
market equity in December of $t-1$.

Book equity is $\mathit{BE} = \mathit{SE} + \mathit{TXDITC} - \mathit{BVPS}$, where
$\mathit{SE}$ is \texttt{SEQ}, else \texttt{CEQ} $+$ \texttt{PSTK}, else \texttt{AT} $-$
\texttt{LT}; $\mathit{TXDITC}$ is \texttt{TXDITC}, or 0 when it is missing; and
$\mathit{BVPS}$ is \texttt{PSTKRV}, else \texttt{PSTKL}, else \texttt{PSTK}, else 0.
Observations with unavailable $\mathit{SE}$ or with $\mathit{BE} \le 0$ are excluded.

The accounting data come from the CRSP/COMPUSTAT Merged file, with link types \texttt{LC}
and \texttt{LU}, primary links (\texttt{LINKPRIM} \texttt{P} or \texttt{C}), the fiscal
year-end date inside the link window, and records reported in US dollars. When a stock has
several records with a fiscal year-end in the same calendar year, a \texttt{P} link is
preferred to a \texttt{C} link, and then the latest fiscal year-end date is used. To avoid
backfilling bias, the record used must be preceded by at least two earlier records
(distinct fiscal year-end dates) of the same firm in the file.

In the release of the CZ Dataset used here (2025.10), \texttt{BMdec} is the book-to-market
ratio in levels and not its logarithm, so $\BM^{CZ} = \texttt{BMdec}$ is used instead of
$\exp(\texttt{BMdec})$. For each month $\tau$, the cross-firm regression
\begin{equation}
  \BM^{CZ}_{j,\tau} = a_\tau + b_\tau\, \BM_{j,\tau} + \varepsilon_{j,\tau}
  \label{eq:q3b_regression}
\end{equation}
is estimated by OLS, with $\BM^{CZ}$ merged on the stock identifier and the month $\tau$,
using the stocks in the universe at $\tau$ for which both signals are non-missing
(1{,}803{,}835 stock-month observations in 739 months; between 454 and 3{,}914 stocks per
month). Figures~\ref{fig:q3b_intercept}--\ref{fig:q3b_r2} plot $\hat a_\tau$, $\hat b_\tau$
and the $R^2$ of these regressions (code: \texttt{src/q3b.py}, \texttt{src/q3\_common.py}).

\subsection{Decile portfolios}
% TODO (Q3c): interpretation of the results (student to write)
% Figures and table produced by src/q3c.py (common rules in src/q3_common.py)
% from spec/q3c.md and spec/q3_common.md.
Decile portfolios are formed on each of the three signals of the CZ Dataset,
$\BM^{CZ} = \texttt{BMdec}$ (in levels, as in Question~3b), $\MOM^{CZ} = \texttt{Mom12m}$ and
$\GP^{CZ} = \texttt{GP}$, dated $\tau$. A stock enters a sort if it is in the universe at the
formation date and its signal is non-missing and finite; for $\BM^{CZ}$, stocks with
$\texttt{BMdec} \le 0$ are excluded.

With annual rebalancing, the portfolios are formed at the end of June of year $t$ with the
signal and the universe of June $t$ and are held for the return months from July of year $t$
to June of year $t+1$; a stock without a signal in June $t$ is left out until the next June.
With monthly rebalancing, the portfolios are formed at the end of month $\tau$ with the signal
and the universe of $\tau$ and are held for the return month $\tau+1$. The breakpoints are the
10th, 20th, \dots, 90th percentiles of the signal at the formation date, computed among the
NYSE stocks in the sort (NYSE breakpoints) or among all stocks in the sort (general
breakpoints); a stock is in decile $d$ if its signal is above the $(d-1)$-th breakpoint and
at or below the $d$-th, so that decile 1 has the lowest and decile 10 the highest signal.

The portfolio return of month $\tau+1$ uses the members with a non-missing return in
$\tau+1$, with equal weights (EW) or with weights proportional to the market equity at the
end of month $\tau$ (VW), which are updated every month, also within the annual holding
period; a member with missing market equity at $\tau$ is excluded from the VW return of
$\tau+1$. Excess returns are $\xR_{p,\tau+1} = R_{p,\tau+1} - R_{f,\tau+1}$, with the
risk-free return from the Ken French data library. The five types of portfolios are:
(i) VW, annual rebalancing, NYSE breakpoints; (ii) EW, annual rebalancing, NYSE breakpoints;
(iii) VW, monthly rebalancing, NYSE breakpoints; (iv) VW, annual rebalancing, general
breakpoints; (v) EW, monthly rebalancing, general breakpoints. The return months are
1963:07--2025:01 (739 months) for all five types; the last annual formation is June 2024
(code: \texttt{src/q3c.py}, \texttt{src/q3\_common.py}).

\subsection{Fama--MacBeth regressions}
% TODO (Q3d): interpretation of the results (student to write)
% Table produced by src/q3d.py (common rules in src/q3_common.py)
% from spec/q3d.md and spec/q3_common.md.
The signals at month $\tau$ are $\BM^{CZ} = \texttt{BMdec}$ and $\GP^{CZ} = \texttt{GP}$ from
the CZ Dataset, as in Question~3c, and the equity duration $\Dur$ from the Dur Dataset, where
the value with \texttt{FF.YEAR} $= t$ is the signal for the months from June of year $t$ to
May of year $t+1$, the timing of $\BM$. A signal is valid if it is non-missing and finite;
for $\BM^{CZ}$, stocks with $\texttt{BMdec} \le 0$ are excluded. The quantile of signal $X$ is
\begin{equation}
  Q^{X}_{j,\tau} = \frac{\mathrm{rank}\left(X_{j,\tau}\right)}{N^{X}_{\tau}},
  \qquad X \in \{\BM^{CZ}, \GP^{CZ}, \Dur\},
  \label{eq:q3d_quantile}
\end{equation}
where the rank (1 for the lowest value, the average rank for ties) and the number of stocks
$N^{X}_{\tau}$ are taken over all stocks in the universe at $\tau$ with a valid $X$, whether
or not they enter a regression.

For each month $\tau$, the following cross-firm regressions of the excess return
$\xR_{j,\tau+1} = R_{j,\tau+1} - R_{f,\tau+1}$, in percent per month, are estimated by OLS and
by WLS with weights equal to the market equity $\mathit{ME}_{j,\tau}$:
\begin{align}
  \text{(i)}\quad   & \xR_{j,\tau+1} = a + b_{\BM}\, Q^{\BM}_{j,\tau} + \varepsilon_{j,\tau+1}, \notag\\
  \text{(ii)}\quad  & \xR_{j,\tau+1} = a + b_{\GP}\, Q^{\GP}_{j,\tau} + \varepsilon_{j,\tau+1}, \notag\\
  \text{(iii)}\quad & \xR_{j,\tau+1} = a + b_{\Dur}\, Q^{\Dur}_{j,\tau} + \varepsilon_{j,\tau+1}, \notag\\
  \text{(iv)}\quad  & \xR_{j,\tau+1} = a + b_{\BM}\, Q^{\BM}_{j,\tau} + b_{\GP}\, Q^{\GP}_{j,\tau}
                      + \varepsilon_{j,\tau+1}, \label{eq:q3d_regressions}\\
  \text{(v)}\quad   & \xR_{j,\tau+1} = a + b_{\Dur}\, Q^{\Dur}_{j,\tau} + b_{\BM}\, Q^{\BM}_{j,\tau}
                      + \varepsilon_{j,\tau+1}, \notag\\
  \text{(vi)}\quad  & \xR_{j,\tau+1} = a + b_{\Dur}\, Q^{\Dur}_{j,\tau} + b_{\GP}\, Q^{\GP}_{j,\tau}
                      + \varepsilon_{j,\tau+1}, \notag\\
  \text{(vii)}\quad & \xR_{j,\tau+1} = a + b_{\Dur}\, Q^{\Dur}_{j,\tau} + b_{\GP}\, Q^{\GP}_{j,\tau}
                      + b_{\BM}\, Q^{\BM}_{j,\tau} + \varepsilon_{j,\tau+1}. \notag
\end{align}
Each regression uses the stocks in the universe at $\tau$ for which all regressors of the
specification are valid and $\xR_{j,\tau+1}$ is non-missing. The signal months are
1963:06--2024:12 for specifications (i), (ii) and (iv), and 1973:06--2024:12 for
specifications (iii), (v), (vi) and (vii), because $\Dur$ starts in 1973. The Fama--MacBeth
estimate is the time-series mean of the monthly coefficient estimates, and its $t$-statistic
comes from a regression of the monthly coefficient series on a constant with Newey--West
(1987, 1994) standard errors, with the lag length selected by the automatic bandwidth rule
as in method (v) of Question~2b. Since $\xR$ is in percent per month, a coefficient is the
difference in percent per month between the highest and the lowest quantile
(code: \texttt{src/q3d.py}, \texttt{src/q3\_common.py}).

\subsection{Yields, forward rates, and returns}
% TODO (Q4a): discussion of the averages (student to write)
% Table produced by src/q4a.py from spec/q4a.md.
From the Fama--Bliss yields $Y^{(H)}_{b,t}$ (in percent) for $H = 1, \dots, 5$, the log yields,
log forward rates and log annual returns are
\begin{equation}
  y^{(H)}_{b,t} = \log\!\left(1 + Y^{(H)}_{b,t}/100\right), \qquad
  f^{(H)}_{b,t} = H\, y^{(H)}_{b,t} - (H-1)\, y^{(H-1)}_{b,t}, \qquad
  r^{(H)}_{b,t} = H\, y^{(H)}_{b,t-1} - (H-1)\, y^{(H-1)}_{b,t},
  \label{eq:q4a_defs}
\end{equation}
with $y^{(0)} = 0$, so that $f^{(1)}_{b,t} = y^{(1)}_{b,t}$ and $r^{(1)}_{b,t} = y^{(1)}_{b,t-1}$;
$t-1$ is 12 months before $t$. The spreads are $\xy^{(H)} = y^{(H)} - y^{(1)}$,
$\xf^{(H)} = f^{(H)} - f^{(1)}$ and $\xr^{(H)} = r^{(H)} - r^{(1)}$
(code: \texttt{src/q4a.py}).

% TODO (Q4b): discussion of the results (student to write)

Table~\ref{tab:q4b} reports OLS estimates of
\begin{equation}
  \frac{1}{H}\,\xr^{(H)}_{b,t\to t+H} = a^{(H)} + b^{(H)}\,\xy^{(H)}_{b,t} + \varepsilon^{(H)}_t,
  \qquad H = 2, \dots, 5,
  \label{eq:q4b_regression}
\end{equation}
where the hold-to-maturity excess return is
\begin{equation}
  \xr^{(H)}_{b,t\to t+H} = \sum_{h=1}^{H} \xr^{(H-h+1)}_{b,t+h}
  = \sum_{h=1}^{H} \left( r^{(H-h+1)}_{b,t+h} - r^{(1)}_{b,t+h} \right),
  \label{eq:q4b_htm}
\end{equation}
with $r$ and $\xr$ as in (\ref{eq:q4a_defs}), $t+h$ being $12h$ months after $t$, and the
$h = H$ term identically zero because $\xr^{(1)}_{b,t} = 0$. Each regression uses every month
$t$ for which all non-zero terms of (\ref{eq:q4b_htm}) are observed, i.e.\ $t + 12(H-1) \le$
2024:12. The $t$-statistics use Hansen--Hodrick (1980) standard errors with $L = 12H - 1$
lags (code: \texttt{src/q4b.py}).
